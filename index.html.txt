    const diffToMon = (day === 0) ? -6 : 1 - day;
    const thisMonday = new Date(now);

function showScreen(id) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    const screen = document.getElementById(id);
    if (screen) screen.classList.add('active');
    document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
    const map = currentMode === 'emp' ? empNavMap : ooNavMap;
    const navId = map[id];
    if (navId) { const el = document.getElementById(navId); if (el) el.classList.add('active'); }
    window.scrollTo(0, 0);
    // Refresh dynamic screens
    if (id === 'emp-dashboard') renderEmpDashboard();
    if (id === 'emp-historial') renderEmpHistorial();
    if (id === 'emp-finances') renderEmpFinances();
    if (id === 'emp-savings-camion') renderEmpCamionDetail();
    if (id === 'oo-dashboard') renderOODashboard();
    if (id === 'oo-stats') renderOOStats();
    if (id === 'oo-historial') renderOOHistorial();
    if (id === 'oo-finances') renderOOFinances();
}

// RENDER — EMPLEADO DASHBOARD
// ════════════════════════════════════════
function renderEmpDashboard() {
    const trips = getTrips().filter(t => t.mode === 'emp');
    const el = document.getElementById('emp-dashboard-content');

    if (trips.length === 0) {
        el.innerHTML = `
            <div class="empty-state">
                <div class="es-icon">🚛</div>
                <div class="es-title">¡Bienvenido a TruckLog Pro!</div>
                <div class="es-sub">Registra tu primer viaje y empieza a construir tu historial. Cada viaje te acerca al camión.</div>
                <button class="btn btn-primary" onclick="showScreen('emp-add-trip')">➕ Registrar primer viaje</button>
            </div>`;
        return;
    }

    const todayStr = today();
    const todayTrips = trips.filter(t => t.date === todayStr);
    const todayTotal = todayTrips.reduce((s,t) => s+t.amount, 0);

    // Yesterday
    const yd = new Date(); yd.setDate(yd.getDate()-1);
    const ydStr = yd.toISOString().split('T')[0];
    const ydTrips = trips.filter(t => t.date === ydStr);
    const ydTotal = ydTrips.reduce((s,t) => s+t.amount, 0);

    // Prev week (por cobrar)
    const prev = getPrevWeekRange();
    const prevTrips = trips.filter(t => t.date >= prev.start && t.date <= prev.end);
    const prevTotal = prevTrips.reduce((s,t) => s+t.amount, 0);
    const prevWait = prevTrips.reduce((s,t) => s+(t.waitTime||0), 0);

    // This month
    const mo = getThisMonthRange();
    const moTrips = trips.filter(t => t.date >= mo.start && t.date <= mo.end);
    const moTotal = moTrips.reduce((s,t) => s+t.amount, 0);
    const moDays = [...new Set(moTrips.map(t=>t.date))].length;

    // Projection
    const daysInMonth = new Date(new Date().getFullYear(), new Date().getMonth()+1, 0).getDate();
    const dayOfMonth = new Date().getDate();
    const projection = moDays > 0 ? Math.round((moTotal / dayOfMonth) * daysInMonth) : 0;

    const showYesterday = ydTotal > 0;
    const showToday = todayTotal > 0;
    const waitHrs = prevWait > 0 ? (prevWait / 60).toFixed(1) + 'h espera' : '';

    let html = '';

    if (showToday) {
        html += `<div class="tap-card" onclick="openEditDayModal('${todayStr}')">
            <div class="tc-label">📅 Hoy — tap para ver</div>
            <div class="tc-amount" style="color:var(--success);">${fmt$(todayTotal)}</div>
            <div class="tc-sub">${todayTrips.length} viajes</div>
        </div>`;
    } else if (showYesterday) {
        html += `<div class="tap-card" onclick="openEditDayModal('${ydStr}')">
            <div class="tc-label">📅 ${fmtDate(ydStr)} (Ayer) — tap para ver</div>
            <div class="tc-amount" style="color:var(--success);">${fmt$(ydTotal)}</div>
            <div class="tc-sub">${ydTrips.length} viajes${waitHrs ? ' • ' + waitHrs : ''}</div>
        </div>`;
    }

    if (prevTotal > 0) {
        const [ps, pe] = [prev.start, prev.end];
        html += `<div class="cobrar-banner" onclick="openCobrarModal()">
            <div>
                <div class="cb-label">💰 Por Cobrar — Sem ${fmtDate(ps).split(' ').slice(1).join(' ')}–${fmtDate(pe).split(' ').slice(1).join(' ')}</div>
                <div class="cb-sub">Semana pasada • tap para ver desglose →</div>
            </div>
            <div style="text-align:right;">
                <div class="cb-amount">${fmt$(prevTotal)}</div>
                <div class="cb-arrow">Ver detalle →</div>
            </div>
        </div>`;
    }

    html += `<div class="stats-grid">
        <div class="stat-card"><h3>Este Mes</h3><div class="value" style="color:var(--success);font-size:22px;">${fmt$(moTotal)}</div><div style="font-size:12px;color:var(--text-secondary);margin-top:4px;">${moDays} días</div></div>
        <div class="stat-card"><h3>Proyección</h3><div class="value" style="color:var(--accent);font-size:22px;">${fmt$(projection)}</div><div style="font-size:12px;color:var(--text-secondary);margin-top:4px;">fin de mes</div></div>
    </div>`;

    html += `<div class="tap-card" onclick="showScreen('emp-historial')" style="display:flex;justify-content:space-between;align-items:center;">
        <div><div class="tc-label">Historial Completo</div><div style="font-size:16px;font-weight:700;">Ver todos mis viajes →</div></div>
        <div style="font-size:28px;">📋</div>
    </div>`;

    el.innerHTML = html;
}

// RENDER — EMPLEADO HISTORIAL
// ════════════════════════════════════════
function renderEmpHistorial() {
    const trips = getTrips().filter(t => t.mode === 'emp');
    const el = document.getElementById('emp-historial-content');

    if (trips.length === 0) {
        el.innerHTML = `<div class="empty-state"><div class="es-icon">📋</div><div class="es-title">Sin historial aún</div><div class="es-sub">Tus viajes aparecerán aquí organizados por mes, semana y día.</div><button class="btn btn-primary" onclick="showScreen('emp-add-trip')">➕ Registrar primer viaje</button></div>`;
        return;
    }

    // Group: month → week → day
    const byMonth = {};
    trips.forEach(t => {
        const mo = t.date.slice(0,7);
        const wk = getWeekKey(t.date);
        if (!byMonth[mo]) byMonth[mo] = {};
        if (!byMonth[mo][wk]) byMonth[mo][wk] = {};
        if (!byMonth[mo][wk][t.date]) byMonth[mo][wk][t.date] = [];
        byMonth[mo][wk][t.date].push(t);
    });

    let html = '';
    const months = Object.keys(byMonth).sort().reverse();
    months.forEach(mo => {
        const weeks = byMonth[mo];
        const moTotal = Object.values(weeks).flatMap(w => Object.values(w).flat()).reduce((s,t)=>s+t.amount,0);
        const moDays = Object.values(weeks).reduce((s,w)=>s+Object.keys(w).length,0);
        html += `<div class="hist-block">
            <div class="hist-header" onclick="toggleAccordion(this)">
                <div class="hist-header-left"><span class="hist-arrow">›</span><div><div class="hist-label">${fmtMonth(mo+'-01')}</div><div class="hist-sub">${moDays} días trabajados</div></div></div>
                <div class="hist-amount">${fmt$(moTotal)}</div>
            </div>
            <div class="hist-children">`;

        Object.keys(weeks).sort().reverse().forEach(wk => {
            const days = weeks[wk];
            const wkTotal = Object.values(days).flat().reduce((s,t)=>s+t.amount,0);
            const wkTrips = Object.values(days).flat().length;
            const wkDays = Object.keys(days).length;
            const wkEnd = new Date(wk); wkEnd.setDate(wkEnd.getDate()+6);
            const wkEndStr = wkEnd.toISOString().split('T')[0];
            html += `<div class="hist-block" style="margin-bottom:6px;">
                <div class="hist-header" onclick="toggleAccordion(this)">
                    <div class="hist-header-left"><span class="hist-arrow">›</span><div><div class="hist-label">Sem ${fmtDate(wk).split(' ').slice(1).join(' ')}–${fmtDate(wkEndStr).split(' ').slice(1).join(' ')}</div><div class="hist-sub">${wkDays} días • ${wkTrips} viajes</div></div></div>
                    <div class="hist-amount">${fmt$(wkTotal)}</div>
                </div>
                <div class="hist-children">`;

            Object.keys(days).sort().reverse().forEach(date => {
                const dayTrips = days[date];
                const dayTotal = dayTrips.reduce((s,t)=>s+t.amount,0);
                html += `<div class="hist-day">
                    <div class="hist-day-header" onclick="toggleDayAcc(this)">
                        <div class="hist-day-left"><span class="hist-arrow" style="font-size:11px;">›</span><div><div class="hist-day-label">${fmtDate(date)}</div><div style="font-size:11px;color:var(--text-secondary);">${dayTrips.length} viajes</div></div></div>
                        <div class="hist-day-right"><div class="hist-day-amount">${fmt$(dayTotal)}</div><div class="edit-day-btn" onclick="event.stopPropagation(); openEditDayModal('${date}')">✏️</div></div>
                    </div>
                    <div class="hist-day-trips">`;
                dayTrips.forEach(t => {
                    const wl = t.waitTime > 0 ? ` • ⏳ ${waitLabel(t.waitTime)}` : '';
                    html += `<div class="hist-trip-row">
                        <div><div>${t.route}</div><div class="hist-trip-note">${t.methodLabel}${wl}${t.notes ? ' · ' + t.notes : ''}</div></div>
                        <div style="display:flex;align-items:center;gap:8px;">
                            <span class="pos">${fmt$(t.amount)}</span>
                            <button class="delete-trip-btn" onclick="askDeleteTrip('${t.id}','emp')" title="Borrar viaje">🗑️</button>
                        </div>
                    </div>`;
                });
                html += `</div></div>`;
            });

            html += `</div></div>`;
        });

        html += `</div></div>`;
    });

    el.innerHTML = html;
}

// RENDER — O/O STATS
// ════════════════════════════════════════
function renderOOStats() {
    const trips = getTrips().filter(t => t.mode === 'oo');
    const ooExp = getOOExpenses();
    const milesData = getMiles();
    const el = document.getElementById('oo-stats-content');

    if (trips.length === 0) {
        el.innerHTML = `<div class="empty-state"><div class="es-icon">📊</div><div class="es-title">Sin datos aún</div><div class="es-sub">Registra viajes para ver tus estadísticas.</div></div>
        <div class="alfred-export-card" onclick="showScreen('consejo-export')">
            <div class="ae-title">🧠 El Consejo de las IAs</div>
            <div class="ae-sub">Exporta tus datos en JSON estructurado para análisis estratégico con Claude o Gemini.</div>
            <button class="ae-btn">📋 Exportar para El Consejo</button>
            <div class="ae-hint">Funciona 100% offline • Copia y pega en cualquier IA</div>
        </div>`;
        return;
    }

    const mo = getThisMonthRange();
    const moTrips = trips.filter(t => t.date >= mo.start && t.date <= mo.end);
    const moMiles = milesData.filter(m => m.date >= mo.start && m.date <= mo.end).reduce((s,m)=>s+m.miles,0);
    const moDays = [...new Set(moTrips.map(t=>t.date))].length;
    const moGross = moTrips.reduce((s,t)=>s+t.amount,0);
    const avgPerDay = moDays > 0 ? Math.round(moGross / moDays) : 0;

    // Best week
    const wkTotals = {};
    trips.forEach(t => { const wk = getWeekKey(t.date); wkTotals[wk] = (wkTotals[wk]||0)+t.amount; });
    const bestWk = Math.max(0, ...Object.values(wkTotals));

    // Clients
    const clientTotals = {};
    trips.forEach(t => { if (t.client) clientTotals[t.client] = (clientTotals[t.client]||0)+t.amount; });
    const clientRows = Object.entries(clientTotals).sort((a,b)=>b[1]-a[1])
        .map(([k,v]) => `<div class="row"><span>${k}</span><span class="pos">${fmt$(v)}</span></div>`).join('');

    el.innerHTML = `
        <div class="stats-grid">
            <div class="stat-card"><h3>🛣️ Millas/Mes</h3><div class="value" style="color:var(--accent);">${moMiles.toLocaleString()}</div></div>
            <div class="stat-card"><h3>💰 Prom/Día</h3><div class="value" style="color:var(--success);">${fmt$(avgPerDay)}</div></div>
            <div class="stat-card"><h3>📈 Mejor Sem.</h3><div class="value">${fmt$(bestWk)}</div></div>
            <div class="stat-card"><h3>🚛 Viajes/Mes</h3><div class="value">${moTrips.length}</div></div>
        </div>
        ${clientRows ? `<div class="card"><h2>👥 Ingresos por Cliente</h2>${clientRows}</div>` : ''}
        <div class="card"><a class="link-btn" onclick="showScreen('oo-historial')" style="margin:0;font-size:16px;">📅 Ver Historial Completo →</a></div>
        <div class="alfred-export-card" onclick="showScreen('consejo-export')">
            <div class="ae-title">🧠 El Consejo de las IAs</div>
            <div class="ae-sub">Exporta tus datos en JSON estructurado. Llévalo a Claude o Gemini para análisis estratégico.</div>
            <button class="ae-btn">📋 Exportar para El Consejo</button>
            <div class="ae-hint">Funciona 100% offline • Copia y pega en cualquier IA</div>
        </div>`;
}

// RENDER — O/O HISTORIAL
// ════════════════════════════════════════
function renderOOHistorial() {
    const trips = getTrips().filter(t => t.mode === 'oo');
    const el = document.getElementById('oo-historial-content');

    if (trips.length === 0) {
        el.innerHTML = `<div class="empty-state"><div class="es-icon">📋</div><div class="es-title">Sin historial aún</div><div class="es-sub">Tus viajes aparecerán aquí.</div></div>`;
        return;
    }

    const byMonth = {};
    trips.forEach(t => {
        const mo = t.date.slice(0,7);
        const wk = getWeekKey(t.date);
        if (!byMonth[mo]) byMonth[mo] = {};
        if (!byMonth[mo][wk]) byMonth[mo][wk] = {};
        if (!byMonth[mo][wk][t.date]) byMonth[mo][wk][t.date] = [];
        byMonth[mo][wk][t.date].push(t);
    });

    let html = '';
    Object.keys(byMonth).sort().reverse().forEach(mo => {
        const weeks = byMonth[mo];
        const moTotal = Object.values(weeks).flatMap(w=>Object.values(w).flat()).reduce((s,t)=>s+t.amount,0);
        const moDays = Object.values(weeks).reduce((s,w)=>s+Object.keys(w).length,0);
        html += `<div class="hist-block">
            <div class="hist-header" onclick="toggleAccordion(this)">
                <div class="hist-header-left"><span class="hist-arrow">›</span><div><div class="hist-label">${fmtMonth(mo+'-01')}</div><div class="hist-sub">${moDays} días</div></div></div>
                <div class="hist-amount">${fmt$(moTotal)}</div>
            </div>
            <div class="hist-children">`;
        Object.keys(weeks).sort().reverse().forEach(wk => {
            const days = weeks[wk];
            const wkTotal = Object.values(days).flat().reduce((s,t)=>s+t.amount,0);
            const wkEnd = new Date(wk); wkEnd.setDate(wkEnd.getDate()+6);
            html += `<div class="hist-block" style="margin-bottom:6px;">
                <div class="hist-header" onclick="toggleAccordion(this)">
                    <div class="hist-header-left"><span class="hist-arrow">›</span><div><div class="hist-label">Sem ${fmtDate(wk).split(' ').slice(1).join(' ')}–${fmtDate(wkEnd.toISOString().split('T')[0]).split(' ').slice(1).join(' ')}</div><div class="hist-sub">${Object.values(days).flat().length} viajes</div></div></div>
                    <div class="hist-amount">${fmt$(wkTotal)}</div>
                </div>
                <div class="hist-children">`;
            Object.keys(days).sort().reverse().forEach(date => {
                const dayTrips = days[date];
                const dayTotal = dayTrips.reduce((s,t)=>s+t.amount,0);
                html += `<div class="hist-day">
                    <div class="hist-day-header" onclick="toggleDayAcc(this)">
                        <div class="hist-day-left"><span class="hist-arrow" style="font-size:11px;">›</span><div><div class="hist-day-label">${fmtDate(date)}</div><div style="font-size:11px;color:var(--text-secondary);">${dayTrips.length} viajes</div></div></div>
                        <div class="hist-day-right"><div class="hist-day-amount">${fmt$(dayTotal)}</div><div class="edit-day-btn" onclick="event.stopPropagation(); openEditDayModal('${date}')">✏️</div></div>
                    </div>
                    <div class="hist-day-trips">`;
                dayTrips.forEach(t => {
                    const statusBadge = t.status === 'paid'
                        ? '<span class="badge badge-paid">✅</span>'
                        : '<span class="badge badge-pending">⏳</span>';
                    html += `<div class="hist-trip-row">
                        <div><div>${t.route}</div><div class="hist-trip-note">${t.methodLabel}${t.client ? ' • ' + t.client : ''}</div></div>
                        <div style="display:flex;align-items:center;gap:6px;">
                            ${statusBadge}
                            <span class="${t.status==='pending'?'':'pos'}" style="${t.status==='pending'?'color:var(--warning);font-weight:600;font-family:\'JetBrains Mono\',monospace;':''}">${fmt$(t.amount)}</span>
                            <button class="delete-trip-btn" onclick="askDeleteTrip('${t.id}','oo')" title="Borrar">🗑️</button>
                        </div>
                    </div>`;
                });
                html += `</div></div>`;
            });
            html += `</div></div>`;
        });
        html += `</div></div>`;
    });
    el.innerHTML = html;
}

    const ooExp = getOOExpenses();
    const goals = getGoals();
    const el = document.getElementById('oo-finances-content');

    const yearTrips = trips;
    const yearGross = yearTrips.reduce((s,t)=>s+t.amount,0);
    const yearExp = ooExp.reduce((s,e)=>s+e.amount,0);
    const yearNeto = yearGross - yearExp;

    const mo = getThisMonthRange();
    const moTrips = trips.filter(t => t.date >= mo.start && t.date <= mo.end);
    const moGross = moTrips.reduce((s,t)=>s+t.amount,0);

            <div class="row total"><span>Ganancia:</span><span class="pos">${fmt$(yearNeto)}</span></div>
        </div>

            <div class="row"><span>Ingresos:</span><span class="pos">${fmt$(moGross)}</span></div>
            ${moExp > 0 ? `<div class="row"><span>Gastos:</span><span class="neg">-${fmt$(moExp)}</span></div>` : ''}

            <div class="section-title" style="margin:0;">💰 Metas de Ahorro</div>

function openEditDayModal(date) {
    currentEditDay = date;
    const trips = getTrips().filter(t => t.date === date);
    const dayTotal = trips.reduce((s,t)=>s+t.amount,0);
    document.getElementById('edit-day-title').textContent = '✏️ ' + fmtDate(date);
    document.getElementById('edit-day-total').textContent = fmt$(dayTotal);

    let html = '';
    if (trips.length === 0) {
        html = '<div style="text-align:center;padding:20px;color:var(--text-secondary);">Sin viajes este día</div>';
    } else {
        trips.forEach(t => {
            const wl = t.waitTime > 0 ? ` • ⏳ ${waitLabel(t.waitTime)}` : '';
            html += `<div class="trip-item">
                <div style="flex:1;"><div class="route">${t.route}</div><div class="method-badge">${t.methodLabel}${wl}</div></div>
                <div class="right">
                    <div class="amount">${fmt$(t.amount)}</div>
                    <button class="delete-trip-btn" onclick="askDeleteTrip('${t.id}','modal')" style="font-size:18px;margin-top:4px;" title="Borrar viaje">🗑️</button>
                </div>
            </div>`;
        });
    }
    document.getElementById('edit-day-trips').innerHTML = html;

    // Set date label for pasado screen
    document.getElementById('pasado-date-label').innerHTML = `📅 Agregando viaje para: <strong>${fmtDate(date)}</strong>`;
    openModal('edit-day-modal');
}

// COBRAR MODAL — EMPLEADO
// ════════════════════════════════════════
function openCobrarModal() {
    const prev = getPrevWeekRange();
    const trips = getTrips().filter(t => t.mode === 'emp' && t.date >= prev.start && t.date <= prev.end);

    // Group by day
    const byDay = {};
    trips.forEach(t => { if (!byDay[t.date]) byDay[t.date] = []; byDay[t.date].push(t); });

    const total = trips.reduce((s,t)=>s+t.amount,0);
    document.getElementById('cobrar-modal-title').textContent = `📅 Sem ${fmtDate(prev.start).split(' ').slice(1).join(' ')} – ${fmtDate(prev.end).split(' ').slice(1).join(' ')}`;

    let html = `<div style="background:var(--accent-light);border-radius:12px;padding:18px;margin-bottom:16px;text-align:center;">
        <div style="font-size:13px;color:var(--accent);font-weight:700;margin-bottom:4px;">TOTAL DE LA SEMANA</div>
        <div style="font-family:'JetBrains Mono',monospace;font-size:32px;font-weight:700;color:var(--accent);">${fmt$(total)}</div>
        <div style="font-size:12px;color:var(--text-secondary);margin-top:4px;">${Object.keys(byDay).length} días • ${trips.length} viajes</div>
    </div>`;
    if (Object.keys(byDay).length > 0) {
        html += `<div style="font-size:13px;color:var(--text-secondary);margin-bottom:8px;font-weight:600;">Desglose por día:</div>`;
        Object.keys(byDay).sort().forEach(d => {
            const dt = byDay[d].reduce((s,t)=>s+t.amount,0);
            html += `<div class="row"><span>${fmtDate(d)}</span><span class="pos">${fmt$(dt)}</span></div>`;
        });
    } else {
        html += `<div style="text-align:center;color:var(--text-secondary);padding:16px;">Sin viajes la semana pasada</div>`;
    }
    document.getElementById('cobrar-modal-content').innerHTML = html;
    openModal('cobrar-modal');
}

// DELETE TRIP
// ════════════════════════════════════════
function askDeleteTrip(id, source) {
    const trips = getTrips();
    const t = trips.find(x=>x.id===id);
    if (!t) return;
    tripToDelete = { id, source };
    document.getElementById('confirm-delete-desc').textContent = `${t.route} — ${fmt$(t.amount)}. Esta acción no se puede deshacer.`;
    openModal('confirm-delete-modal');
}

function confirmDeleteTrip() {
    if (!tripToDelete) return;
    const trips = getTrips();
    const filtered = trips.filter(t => t.id !== tripToDelete.id);
    saveTrips(filtered);
    closeModal('confirm-delete-modal');
    const src = tripToDelete.source;
    tripToDelete = null;
    if (src === 'modal' && currentEditDay) {
        openEditDayModal(currentEditDay);
        renderEmpHistorial();
        renderOOHistorial();
        renderEmpDashboard();
        renderOODashboard();
    } else if (src === 'emp') {
        renderEmpHistorial();
        renderEmpDashboard();
    } else if (src === 'oo') {
        renderOOHistorial();
        renderOODashboard();
    }
}

// EMPLEADO — SAVE TRIP
// ════════════════════════════════════════
let empSessionTrips = []; // trips added this session (before terminar día)

function empSetPayMethod(m) {
    empPayMethod = m;
    ['trip','ton','hour'].forEach(t => {
        document.getElementById('emp-tab-'+t).classList.toggle('active', t===m);
        document.getElementById('emp-method-'+t).style.display = t===m?'block':'none';
    });
    empCalcTrip();
}
function empCalcTrip() {
    let total = 0;
    if (empPayMethod==='trip') total = parseFloat(document.getElementById('emp-pay-flat').value)||0;
    else if (empPayMethod==='ton') { const t=parseFloat(document.getElementById('emp-pay-tons').value)||0; const r=parseFloat(document.getElementById('emp-pay-rate-ton').value)||0; total=t*r; }
    else if (empPayMethod==='hour') { const h=parseFloat(document.getElementById('emp-pay-hours').value)||0; const r=parseFloat(document.getElementById('emp-pay-rate-hour').value)||0; total=h*r; }
    document.getElementById('emp-trip-total-display').textContent = '$'+total.toFixed(2);
}
function empSetWaitTime(mins, el) {
    empWaitTime = mins;
    document.querySelectorAll('#emp-wait-time-selector .wait-btn').forEach(b=>b.classList.remove('active'));
    el.classList.add('active');
}
function empSaveTrip() {
    const route = document.getElementById('emp-trip-route').value.trim();
    if (!route) { alert('Ingresa la ruta'); return; }
    let amount=0, methodLabel='';
    if (empPayMethod==='trip') { amount=parseFloat(document.getElementById('emp-pay-flat').value)||0; methodLabel='Por viaje'; }
    else if (empPayMethod==='ton') { const t=parseFloat(document.getElementById('emp-pay-tons').value)||0; const r=parseFloat(document.getElementById('emp-pay-rate-ton').value)||0; amount=t*r; methodLabel=`${t} ton × $${r}/ton`; }
    else if (empPayMethod==='hour') { const h=parseFloat(document.getElementById('emp-pay-hours').value)||0; const r=parseFloat(document.getElementById('emp-pay-rate-hour').value)||0; amount=h*r; methodLabel=`${h} hrs × $${r}/hr`; }
    if (amount<=0) { alert('Ingresa el pago'); return; }
    const notes = document.getElementById('emp-trip-notes').value.trim();
    const dateVal = document.getElementById('emp-trip-date').value || today();
    const trip = { id:uid(), mode:'emp', date:dateVal, route, amount, methodLabel, waitTime:empWaitTime, notes, status:'paid' };
    const trips = getTrips();
    trips.push(trip);
    saveTrips(trips);
    empSessionTrips.push(trip);
    empRenderSessionTrips();
    // Reset
    document.getElementById('emp-trip-route').value='';
    document.getElementById('emp-trip-notes').value='';
    document.getElementById('emp-pay-flat').value='';
    document.getElementById('emp-trip-total-display').textContent='$0.00';
    empWaitTime=0;
    document.querySelectorAll('#emp-wait-time-selector .wait-btn').forEach(b=>b.classList.remove('active'));
    document.querySelector('#emp-wait-time-selector .wait-btn').classList.add('active');
}
function empRenderSessionTrips() {
    const box = document.getElementById('emp-trips-today-box');
    const list = document.getElementById('emp-trips-today-list');
    const totalEl = document.getElementById('emp-trips-today-total');
    if (empSessionTrips.length===0) { box.style.display='none'; return; }
    box.style.display='block';
    let total=0, html='';
    empSessionTrips.forEach(t => {
        total+=t.amount;
        const wl = t.waitTime>0?` • ⏳ ${waitLabel(t.waitTime)}`:'';
        const nl = t.notes?`<div style="font-size:11px;color:var(--text-secondary);margin-top:2px;">📝 ${t.notes}</div>`:'';
        html+=`<div class="trip-item"><div><div class="route">${t.route}</div><div class="method-badge">${t.methodLabel}${wl}</div>${nl}</div><div class="right"><div class="amount">${fmt$(t.amount)}</div></div></div>`;
    });
    list.innerHTML=html;
    totalEl.innerHTML=`<span>Total (${empSessionTrips.length} viajes):</span><span class="pos">${fmt$(total)}</span>`;
}
function empFinishDay() {
    if (empSessionTrips.length===0) { alert('No hay viajes registrados hoy'); return; }
    const total = empSessionTrips.reduce((s,t)=>s+t.amount,0);
    alert(`✅ Día terminado!\n${empSessionTrips.length} viajes • ${fmt$(total)} total`);
    empSessionTrips=[];
    empRenderSessionTrips();
    showScreen('emp-dashboard');
}

function calcTrip() {
    let total=0;
    if (payMethod==='trip') total=parseFloat(document.getElementById('pay-flat').value)||0;
    else if (payMethod==='ton') { const t=parseFloat(document.getElementById('pay-tons').value)||0; const r=parseFloat(document.getElementById('pay-rate-ton').value)||0; total=t*r; }
    else if (payMethod==='hour') { const h=parseFloat(document.getElementById('pay-hours').value)||0; const r=parseFloat(document.getElementById('pay-rate-hour').value)||0; total=h*r; }
    document.getElementById('trip-total-display').textContent='$'+total.toFixed(2);
}

}
function setWaitTime(mins, el) {

function saveOOTrip() {
    const route=document.getElementById('trip-route').value.trim();
    if (!route){alert('Ingresa la ruta');return;}
    let amount=0, methodLabel='';
    if (payMethod==='trip'){amount=parseFloat(document.getElementById('pay-flat').value)||0;methodLabel='Por viaje';}
    else if (payMethod==='ton'){const t=parseFloat(document.getElementById('pay-tons').value)||0;const r=parseFloat(document.getElementById('pay-rate-ton').value)||0;amount=t*r;methodLabel=`${t} ton × $${r}/ton`;}
    else if (payMethod==='hour'){const h=parseFloat(document.getElementById('pay-hours').value)||0;const r=parseFloat(document.getElementById('pay-rate-hour').value)||0;amount=h*r;methodLabel=`${h} hrs × $${r}/hr`;}
    if (amount<=0){alert('Ingresa el pago');return;}
    const client=document.getElementById('trip-client').value.trim();
    const notes=document.getElementById('trip-notes').value.trim();
    const dateVal=document.getElementById('trip-date').value||today();
    const trip={id:uid(),mode:'oo',date:dateVal,route,amount,methodLabel,waitTime:waitTimeMinutes,notes,status:payStatus,client};
    const trips=getTrips();trips.push(trip);saveTrips(trips);
    ooSessionTrips.push(trip);
    renderOOSessionTrips();
    document.getElementById('trip-route').value='';
    document.getElementById('trip-notes').value='';
    document.getElementById('trip-client').value='';
    document.getElementById('pay-flat').value='';
    document.getElementById('trip-total-display').textContent='$0.00';
    waitTimeMinutes=0;
    document.querySelectorAll('#wait-time-selector .wait-btn').forEach(b=>b.classList.remove('active'));
    document.querySelector('#wait-time-selector .wait-btn').classList.add('active');
    document.getElementById('wait-time-info').style.display='none';
    setStatus('paid');
}
function renderOOSessionTrips() {
    const box=document.getElementById('trips-today-box');
    const list=document.getElementById('trips-today-list');
    const totalEl=document.getElementById('trips-today-total');
    if (ooSessionTrips.length===0){box.style.display='none';return;}
    box.style.display='block';
    let total=0,html='';
    ooSessionTrips.forEach(t=>{
        total+=t.amount;
        const badge=t.status==='paid'?'<span class="badge badge-paid">✅</span>':'<span class="badge badge-pending">⏳</span>';
        html+=`<div class="trip-item"><div><div class="route">${t.route}</div><div class="method-badge">${t.methodLabel}</div></div><div class="right"><div class="amount">${fmt$(t.amount)}</div>${badge}</div></div>`;
    });
    list.innerHTML=html;
    totalEl.innerHTML=`<span>Total (${ooSessionTrips.length} viajes):</span><span class="pos">${fmt$(total)}</span>`;
}
function ooFinishDay() {
    if (ooSessionTrips.length===0){alert('No hay viajes registrados hoy');return;}
    const total=ooSessionTrips.reduce((s,t)=>s+t.amount,0);
    alert(`✅ Día terminado!\n${ooSessionTrips.length} viajes • ${fmt$(total)} total`);
    ooSessionTrips=[];
    renderOOSessionTrips();
    showScreen('oo-dashboard');
}

function checkOilWarning() {
    const settings=getSettings();

    const newTotal=settings.currentMiles+miles;
    const remaining=OIL_LIMIT-newTotal;
    const warning=document.getElementById('miles-oil-warning');

    else if (remaining<=0){warning.style.display='block';warning.textContent=`🚨 ¡Ya necesitas cambio de aceite! Pasaste por ${Math.abs(remaining).toFixed(0)} millas`;}
    else warning.style.display='none';
}

function addDeposit(goalKey) {
    const amt = parseFloat(prompt(`Monto del depósito para ${goalKey === 'ambar' ? 'Ámbar' : goalKey === 'camion' ? 'Camión' : goalKey === 'emergencia' ? 'Emergencia' : 'Casa'} ($):`));
    if (!amt || amt <= 0) return;
    const goals = getGoals();
    goals[goalKey].ahorrado = (goals[goalKey].ahorrado || 0) + amt;
    saveGoals(goals);
    if (currentMode === 'emp') { renderEmpFinances(); renderEmpCamionDetail(); }
    else { renderOOFinances(); }
    alert(`✅ Depósito de ${fmt$(amt)} registrado`);
}

function generateEmpJSON() {
    const period=document.getElementById('emp-export-select').value;
    const trips=getTrips().filter(t=>t.mode==='emp');
    const expenses=getExpenses();
    const goals=getGoals();
    let filtered=trips;
    if (period==='week'){const r=getThisWeekRange();filtered=trips.filter(t=>t.date>=r.start&&t.date<=r.end);}
    else if (period==='month'){const r=getThisMonthRange();filtered=trips.filter(t=>t.date>=r.start&&t.date<=r.end);}
    const data={
        exportado_por:"TruckLog Pro V4.5 — Modo Empleado",
        fecha_exportacion:today(),
        periodo:period,
        resumen:{total_ganado:filtered.reduce((s,t)=>s+t.amount,0),viajes:filtered.length,dias_trabajados:[...new Set(filtered.map(t=>t.date))].length},

    document.getElementById('emp-copy-json-btn').style.display='block';
    document.getElementById('emp-consejo-instructions').style.display='block';

}
